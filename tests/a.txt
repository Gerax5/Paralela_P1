#include <stdlib.h>
#include <math.h>
#include <omp.h>
#include "lloyd.h"
#include "image.h"
#include "stippling.h"
#include "voronoi.h"
#include "config.h"

static inline unsigned lcg(unsigned *st) {
  *st = (*st * 1664525u + 1013904223u);
  return *st;
}

static inline int irand_range(unsigned *st, int hi) {
  return (int)(lcg(st) % (unsigned)hi);
}

bool lloydStep(const Image *img, Stippling *s, int W, int H, int step, float gamma) {
  if (!s || !s->pts || s->count <= 0 || W <= 0 || H <= 0) return false;
  if (!img || !img->pixels) return false;
  if (step < 1) step = 1;

  UniformGrid g = (UniformGrid){0};
  gridBuild(&g, W, H, 32, s);

  const int n = s->count;

  double *sumX = (double *)calloc((size_t)n, sizeof(double));
  double *sumY = (double *)calloc((size_t)n, sizeof(double));
  double *sumW = (double *)calloc((size_t)n, sizeof(double));
  if (!sumX || !sumY || !sumW) {
    free(sumX); free(sumY); free(sumW);
    gridFree(&g);
    return false;
  }

  int useOmp = 0;
  const char *env = getenv("STIPPLE_PARALLEL");
  if (env && *env == '1') useOmp = 1;

  if (!useOmp) {
    for (int y = 0; y < H; y += step) {
      float v = ((float)y + 0.5f) / (float)H;
      for (int x = 0; x < W; x += step) {
        float u = ((float)x + 0.5f) / (float)W;
        float lum = sampleIntensityBilinearUV(img, u, v);
        double w = pow(fmax(0.0, 1.0 - (double)lum), (double)gamma);
        if (w <= 0.0) continue;

        int best = gridNearest(&g, s, (float)x, (float)y, 2);
        if (best < 0) {
          best = 0; float bestD2 = 1e30f;
          for (int i = 0; i < n; i++) {
            float dx = (float)x - s->pts[i].x;
            float dy = (float)y - s->pts[i].y;
            float d2 = dx*dx + dy*dy;
            if (d2 < bestD2) { bestD2 = d2; best = i; }
          }
        }

        sumX[best] += (double)x * w;
        sumY[best] += (double)y * w;
        sumW[best] += w;
      }
    }

    for (int i = 0; i < n; i++) {
      if (sumW[i] > 0.0) {
        s->pts[i].x = (float)(sumX[i] / sumW[i]);
        s->pts[i].y = (float)(sumY[i] / sumW[i]);
      } else {
        unsigned st = (unsigned)(n ^ W ^ H ^ (i * 2654435761u));
        for (int t = 0; t < 64; ++t) {
          int rx = irand_range(&st, W);
          int ry = irand_range(&st, H);
          float u = ((float)rx + 0.5f) / (float)W;
          float v = ((float)ry + 0.5f) / (float)H;
          float lum = sampleIntensityBilinearUV(img, u, v);
          double ww = pow(fmax(0.0, 1.0 - (double)lum), (double)gamma);
          if (ww > 1e-6) { s->pts[i].x = (float)rx; s->pts[i].y = (float)ry; break; }
        }
      }
    }
  } else {
    const int T = omp_get_max_threads();
    double *sumXlocal = (double *)calloc((size_t)T * n, sizeof(double));
    double *sumYlocal = (double *)calloc((size_t)T * n, sizeof(double));
    double *sumWlocal = (double *)calloc((size_t)T * n, sizeof(double));
    if (!sumXlocal || !sumYlocal || !sumWlocal) {
      free(sumXlocal); free(sumYlocal); free(sumWlocal);
      free(sumX); free(sumY); free(sumW);
      gridFree(&g);
      return false;
    }

    #pragma omp parallel
    {
      int tid = omp_get_thread_num();
      double *sx = sumXlocal + (size_t)tid * (size_t)n;
      double *sy = sumYlocal + (size_t)tid * (size_t)n;
      double *sw = sumWlocal + (size_t)tid * (size_t)n;

      #pragma omp for collapse(2) schedule(static,64)
      for (int y = 0; y < H; y += step) {
        for (int x = 0; x < W; x += step) {
          float u = ((float)x + 0.5f) / (float)W;
          float v = ((float)y + 0.5f) / (float)H;
          float lum = sampleIntensityBilinearUV(img, u, v);
          double w = pow(fmax(0.0, 1.0 - (double)lum), (double)gamma);
          if (w <= 0.0) continue;

          int sidx = gridNearest(&g, s, (float)x, (float)y, 2);
          if (sidx < 0) {
            float bestD2 = 1e30f; sidx = 0;
            for (int i = 0; i < n; ++i) {
              float dx = (float)x - s->pts[i].x;
              float dy = (float)y - s->pts[i].y;
              float d2 = dx*dx + dy*dy;
              if (d2 < bestD2) { bestD2 = d2; sidx = i; }
            }
          }

          sx[sidx] += (double)x * w;
          sy[sidx] += (double)y * w;
          sw[sidx] += w;
        }
      }

      #pragma omp for schedule(static)
      for (int i = 0; i < n; ++i) {
        double rx = 0.0, ry = 0.0, rw = 0.0;
        for (int t = 0; t < T; ++t) {
          size_t base = (size_t)t * (size_t)n + (size_t)i;
          rx += sumXlocal[base];
          ry += sumYlocal[base];
          rw += sumWlocal[base];
        }
        if (rw > 0.0) {
          s->pts[i].x = (float)(rx / rw);
          s->pts[i].y = (float)(ry / rw);
        } else {
          unsigned st = (unsigned)(n ^ W ^ H ^ (i * 2654435761u));
          for (int tt = 0; tt < 64; ++tt) {
            int rxp = irand_range(&st, W);
            int ryp = irand_range(&st, H);
            float u = ((float)rxp + 0.5f) / (float)W;
            float v = ((float)ryp + 0.5f) / (float)H;
            float lum = sampleIntensityBilinearUV(img, u, v);
            double ww = pow(fmax(0.0, 1.0 - (double)lum), (double)gamma);
            if (ww > 1e-6) { s->pts[i].x = (float)rxp; s->pts[i].y = (float)ryp; break; }
          }
        }
      }
    }

    free(sumXlocal);
    free(sumYlocal);
    free(sumWlocal);
  }

  free(sumX);
  free(sumY);
  free(sumW);
  gridFree(&g);
  return true;
}
